Create package.json ```bash cat > package.json << 'EOF' { "name": "community-mansion-api", "version": "1.0.0", "description": "Animated Frame Community Mansion with Twitter Integration", "main": "server.js", "scripts": { "start": "node server.js", "dev": "nodemon server.js" }, "dependencies": { "express": "^4.18.2", "cors": "^2.8.5", "sqlite3": "^5.1.6", "express-rate-limit": "^6.7.0", "helmet": "^6.1.5", "dotenv": "^16.0.3", "axios": "^1.4.0", "twitter-api-v2": "^1.15.0", "multer": "^1.4.5" }, "devDependencies": { "nodemon": "^2.0.22" } } EOF ``` #### 2. Create .env file ```bash cat > .env << 'EOF' # Server Configuration PORT=3001 NODE_ENV=production DATABASE_PATH=/opt/community-mansion/mansion.db # Twitter API Credentials - GET NEW ONES from developer.twitter.com TWITTER_API_KEY=your_new_api_key_here TWITTER_API_SECRET=your_new_api_secret_here TWITTER_BEARER_TOKEN=your_new_bearer_token_here TWITTER_ACCESS_TOKEN=your_new_access_token_here TWITTER_ACCESS_TOKEN_SECRET=your_new_access_token_secret_here # CORS Configuration CORS_ORIGIN=https://aabis5004.github.io # Rate Limiting API_RATE_LIMIT=200 TWITTER_RATE_LIMIT=50 # Security SECRET_KEY=your_secret_key_here_change_this EOF ``` #### 3. Create server.js ```bash cat > server.js << 'EOF' require('dotenv').config(); const express = require('express'); const cors = require('cors'); const helmet = require('helmet'); const sqlite3 = require('sqlite3').verbose(); const path = require('path'); const rateLimit = require('express-rate-limit'); const { TwitterApi } = require('twitter-api-v2'); const app = express(); const PORT = process.env.PORT || 3001; // Security middleware app.use(helmet({ crossOriginEmbedderPolicy: false, contentSecurityPolicy: { directives: { defaultSrc: ["'self'"], styleSrc: ["'self'", "'unsafe-inline'", "https://fonts.googleapis.com"], fontSrc: ["'self'", "https://fonts.gstatic.com"], imgSrc: ["'self'", "data:", "https:", "http:"], scriptSrc: ["'self'", "'unsafe-inline'"], connectSrc: ["'self'", "https:"] } } })); // CORS configuration app.use(cors({ origin: [ process.env.CORS_ORIGIN, 'http://localhost:3000', 'http://127.0.0.1:3000', 'https://aabis5004.github.io' ], credentials: true, methods: ['GET', 'POST', 'PUT', 'DELETE', 'OPTIONS'], allowedHeaders: ['Content-Type', 'Authorization', 'X-Requested-With'] })); app.use(express.json({ limit: '10mb' })); app.use(express.urlencoded({ extended: true })); // Rate limiting const generalLimiter = rateLimit({ windowMs: 15 * 60 * 1000, // 15 minutes max: parseInt(process.env.API_RATE_LIMIT) || 200, message: { error: 'Too many requests, please try again later.' } }); const twitterLimiter = rateLimit({ windowMs: 15 * 60 * 1000, // 15 minutes max: parseInt(process.env.TWITTER_RATE_LIMIT) || 50, message: { error: 'Twitter search rate limit exceeded. Please try again later.' } }); app.use('/api/', generalLimiter); app.use('/api/twitter-search', twitterLimiter); // Twitter API setup let twitterClient = null; if (process.env.TWITTER_API_KEY && process.env.TWITTER_API_SECRET) { try { twitterClient = new TwitterApi({ appKey: process.env.TWITTER_API_KEY, appSecret: process.env.TWITTER_API_SECRET, accessToken: process.env.TWITTER_ACCESS_TOKEN, accessSecret: process.env.TWITTER_ACCESS_TOKEN_SECRET, }); } catch (error) { console.error('Twitter API setup failed:', error); } } // Database setup const dbPath = process.env.DATABASE_PATH || path.join(__dirname, 'mansion.db'); const db = new sqlite3.Database(dbPath); // Initialize database db.serialize(() => { // Members table db.run(`CREATE TABLE IF NOT EXISTS members ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT NOT NULL, username TEXT, email TEXT, role TEXT CHECK(role IN ('founder', 'moderator', 'resident', 'guest')) NOT NULL DEFAULT 'guest', source TEXT CHECK(source IN ('manual', 'twitter', 'search')) NOT NULL DEFAULT 'manual', followers INTEGER DEFAULT 0, verified BOOLEAN DEFAULT 0, twitter_id TEXT, profile_image TEXT, description TEXT, created_at DATETIME DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME DEFAULT CURRENT_TIMESTAMP )`); // Mansion residents table db.run(`CREATE TABLE IF NOT EXISTS mansion ( id INTEGER PRIMARY KEY AUTOINCREMENT, member_id INTEGER, name TEXT NOT NULL, role TEXT NOT NULL, room TEXT NOT NULL, source TEXT CHECK(source IN ('manual', 'twitter', 'search')) NOT NULL, joined_at DATE DEFAULT CURRENT_DATE, FOREIGN KEY(member_id) REFERENCES members(id) )`); // Search history db.run(`CREATE TABLE IF NOT EXISTS search_history ( id INTEGER PRIMARY KEY AUTOINCREMENT, query TEXT NOT NULL, source TEXT CHECK(source IN ('twitter', 'manual')) NOT NULL, results_count INTEGER DEFAULT 0, searched_at DATETIME DEFAULT CURRENT_TIMESTAMP, ip_address TEXT )`); // API logs db.run(`CREATE TABLE IF NOT EXISTS api_logs ( id INTEGER PRIMARY KEY AUTOINCREMENT, endpoint TEXT NOT NULL, method TEXT NOT NULL, ip_address TEXT, user_agent TEXT, response_status INTEGER, response_time INTEGER, created_at DATETIME DEFAULT CURRENT_TIMESTAMP )`); }); // Helper functions const dbAsync = { all: (query, params) => new Promise((resolve, reject) => { db.all(query, params, (err, rows) => { if (err) reject(err); else resolve(rows); }); }), run: (query, params) => new Promise((resolve, reject) => { db.run(query, params, function(err) { if (err) reject(err); else resolve({ id: this.lastID, changes: this.changes }); }); }), get: (query, params) => new Promise((resolve, reject) => { db.get(query, params, (err, row) => { if (err) reject(err); else resolve(row); }); }) }; // Logging middleware app.use((req, res, next) => { const start = Date.now(); res.on('finish', () => { const duration = Date.now() - start; // Log API usage (async, don't block response) dbAsync.run( 'INSERT INTO api_logs (endpoint, method, ip_address, user_agent, response_status, response_time) VALUES (?, ?, ?, ?, ?, ?)', [req.path, req.method, req.ip, req.get('User-Agent'), res.statusCode, duration] ).catch(err => console.error('Logging error:', err)); }); next(); }); // API Routes // Health check app.get('/api/health', (req, res) => { res.json({ status: 'OK', service: 'Community Mansion API', version: '2.0.0', timestamp: new Date().toISOString(), features: { twitter: !!twitterClient, database: true, cors: true, rateLimit: true }, uptime: process.uptime() }); }); // Get all members app.get('/api/members', async (req, res) => { try { const members = await dbAsync.all('SELECT * FROM members ORDER BY created_at DESC'); res.json(members); } catch (error) { console.error('Error fetching members:', error); res.status(500).json({ error: 'Failed to fetch members' }); } }); // Add new member app.post('/api/members', async (req, res) => { try { const { name, username, email, role = 'guest', source = 'manual', followers = 0, verified = false, twitter_id, profile_image, description } = req.body; if (!name || name.trim().length === 0) { return res.status(400).json({ error: 'Name is required' }); } // Generate email if not provided const finalEmail = email || `${(username || name).toLowerCase().replace(/\s+/g, '.')}@mansion.com`; const result = await dbAsync.run( `INSERT INTO members (name, username, email, role, source, followers, verified, twitter_id, profile_image, description) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`, [name.trim(), username, finalEmail, role, source, followers, verified, twitter_id, profile_image, description] ); const newMember = await dbAsync.get('SELECT * FROM members WHERE id = ?', [result.id]); res.status(201).json(newMember); } catch (error) { if (error.code === 'SQLITE_CONSTRAINT') { return res.status(409).json({ error: 'Member with this email already exists' }); } console.error('Error adding member:', error); res.status(500).json({ error: 'Failed to add member' }); } }); // Twitter search endpoint app.get('/api/twitter-search', async (req, res) => { try { const { q } = req.query; if (!q || q.trim().length === 0) { return res.status(400).json({ error: 'Search query is required' }); } if (!twitterClient) { // Return mock data if Twitter API not configured const mockResults = [{ name: `${q} Seismic Chad`, username: `${q.toLowerCase().replace(/\s+/g, '_')}_seismic`, email: `${q.toLowerCase().replace(/\s+/g, '.')}@twitter.com`, role: 'guest', source: 'twitter', followers: Math.floor(Math.random() * 10000), verified: Math.random() > 0.8, description: `Seismic researcher and ${q} expert`, profile_image: null }]; // Log the search await dbAsync.run( 'INSERT INTO search_history (query, source, results_count, ip_address) VALUES (?, ?, ?, ?)', [q, 'twitter', mockResults.length, req.ip] ); return res.json(mockResults); } // Log search await dbAsync.run( 'INSERT INTO search_history (query, source, ip_address) VALUES (?, ?, ?)', [q, 'twitter', req.ip] ); // Search Twitter const searchResults = await twitterClient.readOnly.v2.searchUsers(q, { 'user.fields': [ 'public_metrics', 'verified', 'profile_image_url', 'description', 'location', 'url' ], max_results: 10 }); const users = searchResults.data || []; // Format results const formattedUsers = users.map(user => ({ name: user.name, username: user.username, email: `${user.username}@twitter.com`, role: 'guest', source: 'twitter', followers: user.public_metrics?.followers_count || 0, verified: user.verified || false, twitter_id: user.id, profile_image: user.profile_image_url, description: user.description || '', location: user.location, url: user.url })); // Update search results count await dbAsync.run( 'UPDATE search_history SET results_count = ? WHERE query = ? AND source = ? AND ip_address = ? ORDER BY searched_at DESC LIMIT 1', [formattedUsers.length, q, 'twitter', req.ip] ); res.json(formattedUsers); } catch (error) { console.error('Twitter search error:', error); // Return mock data on error const mockResults = [{ name: `${req.query.q} Chad`, username: `${req.query.q.toLowerCase().replace(/\s+/g, '_')}_seismic`, email: `${req.query.q.toLowerCase().replace(/\s+/g, '.')}@twitter.com`, role: 'guest', source: 'twitter', followers: Math.floor(Math.random() * 10000), verified: Math.random() > 0.8, description: `Seismic enthusiast and ${req.query.q} expert` }]; res.json(mockResults); } }); // Get mansion residents app.get('/api/mansion', async (req, res) => { try { const residents = await dbAsync.all('SELECT * FROM mansion ORDER BY joined_at DESC'); res.json(residents); } catch (error) { console.error('Error fetching mansion residents:', error); res.status(500).json({ error: 'Failed to fetch mansion residents' }); } }); // Add member to mansion app.post('/api/mansion', async (req, res) => { try { const { memberId, name, role, room, source } = req.body; if (!name || !role || !source) { return res.status(400).json({ error: 'Name, role, and source are required' }); } // Generate room if not provided const rooms = ['penthouse', 'library', 'garden', 'workshop', 'observatory', 'wine_cellar', 'study', 'lounge']; const finalRoom = room || rooms[Math.floor(Math.random() * rooms.length)]; // Check if member already in mansion const existing = await dbAsync.get( 'SELECT id FROM mansion WHERE name = ?', [name] ); if (existing) { return res.status(409).json({ error: 'Member already in mansion' }); } const result = await dbAsync.run( 'INSERT INTO mansion (member_id, name, role, room, source) VALUES (?, ?, ?, ?, ?)', [memberId, name, role, finalRoom, source] ); const newResident = await dbAsync.get('SELECT * FROM mansion WHERE id = ?', [result.id]); res.status(201).json(newResident); } catch (error) { console.error('Error adding to mansion:', error); res.status(500).json({ error: 'Failed to add to mansion' }); } }); // Remove from mansion app.delete('/api/mansion/:id', async (req, res) => { try { const { id } = req.params; const result = await dbAsync.run('DELETE FROM mansion WHERE id = ?', [id]); if (result.changes === 0) { return res.status(404).json({ error: 'Resident not found' }); } res.json({ message: 'Member removed from mansion', id: parseInt(id) }); } catch (error) { console.error('Error removing from mansion:', error); res.status(500).json({ error: 'Failed to remove from mansion' }); } }); // Statistics endpoint app.get('/api/stats', async (req, res) => { try { const [ totalMembers, mansionResidents, twitterMembers, manualMembers, totalSearches, recentSearches ] = await Promise.all([ dbAsync.get('SELECT COUNT(*) as count FROM members'), dbAsync.get('SELECT COUNT(*) as count FROM mansion'), dbAsync.get('SELECT COUNT(*) as count FROM members WHERE source = "twitter"'), dbAsync.get('SELECT COUNT(*) as count FROM members WHERE source = "manual"'), dbAsync.get('SELECT COUNT(*) as count FROM search_history'), dbAsync.all('SELECT query, searched_at FROM search_history ORDER BY searched_at DESC LIMIT 10') ]); const roleStats = await Promise.all([ dbAsync.get('SELECT COUNT(*) as count FROM members WHERE role = "founder"'), dbAsync.get('SELECT COUNT(*) as count FROM members WHERE role = "moderator"'), dbAsync.get('SELECT COUNT(*) as count FROM members WHERE role = "resident"'), dbAsync.get('SELECT COUNT(*) as count FROM members WHERE role = "guest"') ]); res.json({ members: { total: totalMembers.count, twitter: twitterMembers.count, manual: manualMembers.count, inMansion: mansionResidents.count }, roles: { founders: roleStats[0].count, moderators: roleStats[1].count, residents: roleStats[2].count, guests: roleStats[3].count }, activity: { totalSearches: totalSearches.count, recentSearches: recentSearches }, server: { uptime: process.uptime(), timestamp: new Date().toISOString() } }); } catch (error) { console.error('Error fetching stats:', error); res.status(500).json({ error: 'Failed to fetch statistics' }); } }); // Error handling middleware app.use((err, req, res, next) => { console.error('Unhandled error:', err); res.status(500).json({ error: 'Internal server error' }); }); // 404 handler app.use((req, res) => { res.status(404).json({ error: 'Endpoint not found' }); }); // Graceful shutdown process.on('SIGINT', () => { console.log('\n🛑 Shutting down gracefully...'); db.close((err) => { if (err) { console.error('❌ Error closing database:', err); } else { console.log('✅ Database connection closed.'); } process.exit(0); }); }); // Start server app.listen(PORT, () => { console.log(`🏛️ Community Mansion API running on port ${PORT}`); console.log(`📍 Health: http://localhost:${PORT}/api/health`); console.log(`🐦 Twitter: ${twitterClient ? '✅ ENABLED' : '❌ DISABLED'}`); console.log(`💾 Database: ${dbPath}`); console.log(`🌐 CORS: ${process.env.CORS_ORIGIN}`); console.log(`🚀 Environment: ${process.env.NODE_ENV}`); }); EOF ``` ### Install Dependencies and Setup PM2 ```bash # Install packages npm install # Install PM2 globally npm install -g pm2 # Create PM2 config cat > ecosystem.config.js << 'EOF' module.exports = { apps: [{ name: 'mansion-api', script: 'server.js', instances: 1, autorestart: true, watch: false, max_memory_restart: '1G', env: { NODE_ENV: 'production', PORT: 3001 }, error_file: './logs/err.log', out_file: './logs/out.log', log_file: './logs/combined.log', time: true }] }; EOF # Create logs directory mkdir logs # Start with PM2 pm2 start ecosystem.config.js pm2 save pm2 startup ``` --- ## 🐦 **STEP 3: Twitter API Setup** ### Get New Twitter API Keys (CRITICAL!) 1. **Go to:** https://developer.twitter.com/en/portal/dashboard 2. **Create new app** or regenerate keys for existing app 3. **Get these credentials:** - API Key - API Secret Key - Bearer Token - Access Token - Access Token Secret ### Update Backend Configuration ```bash # Edit .env file nano /opt/community-mansion/.env # Replace placeholders with your ACTUAL new keys: TWITTER_API_KEY=your_actual_api_key_here TWITTER_API_SECRET=your_actual_api_secret_here TWITTER_BEARER_TOKEN=your_actual_bearer_token_here TWITTER_ACCESS_TOKEN=your_actual_access_token_here TWITTER_ACCESS_TOKEN_SECRET=your_actual_access_token_secret_here # Save and exit (Ctrl+X, Y, Enter) ``` ### Restart Backend ```bash pm2 restart mansion-api pm2 logs mansion-api ``` --- ## 🌐 **STEP 4: Setup Nginx & Domain** ### Configure Nginx ```bash cat > /etc/nginx/sites-available/mansion << 'EOF' server { listen 80;
